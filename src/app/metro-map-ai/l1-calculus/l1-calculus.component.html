🤔 ¿Por qué Cálculo para IA?

Entrenar una red neuronal significa minimizar la función de pérdida. ¿Cómo? Calculando en qué dirección bajar más rápido. Ese cálculo ES el cálculo diferencial.

⛰️

Minimización

Encontrar los pesos W que minimizan L(W) = pérdida. El mínimo de una función está donde f'(x) = 0.

∇

Backpropagation

Regla de la cadena aplicada hacia atrás. Calcula cómo contribuye cada peso al error final.

📈

Curva de aprendizaje

La derivada segunda (curvatura) indica si estamos en un mínimo local, máximo o punto de silla.

🎯

Regularización

L1/L2 regularización modifica el gradiente. L1 produce sparsidad, L2 shrinkage uniforme.

🔄

Transformaciones

Softmax, sigmoid, ReLU — todas son funciones diferenciables. Backprop requiere que lo sean.

⚡

Optimizadores

SGD, Momentum, Adam — todos son variantes de gradient descent que usan cálculo diferencial.

📐 Derivadas: La Tasa de Cambio

La derivada f'(x) dice cuánto cambia f cuando x cambia infinitesimalmente. En ML: ¿cuánto cambia la pérdida cuando cambio un peso?

import numpy as np

# ── Derivada numérica (aproximación) ─────────────
def derivada_numerica(f, x, h=1e-7):
    """Aproximación: f'(x) ≈ [f(x+h) - f(x-h)] / 2h"""
    return (f(x + h) - f(x - h)) / (2 * h)

# Funciones de activación comunes
def relu(x):    return np.maximum(0, x)
def sigmoid(x): return 1 / (1 + np.exp(-x))
def tanh(x):    return np.tanh(x)

# Derivadas analíticas
def d_relu(x):    return (x > 0).astype(float)
def d_sigmoid(x): s = sigmoid(x); return s * (1 - s)
def d_tanh(x):    return 1 - np.tanh(x)**2

# Compara numérica vs analítica
x_test = np.array([-2.0, -1.0, 0.0, 1.0, 2.0])
for name, f, df in [('sigmoid', sigmoid, d_sigmoid), ('tanh', tanh, d_tanh)]:
    num  = derivada_numerica(f, x_test)
    anal = df(x_test)
    max_err = np.abs(num - anal).max()
    print(f"{name}: error máx entre numérica y analítica = {max_err:.2e}")

# Reglas básicas
print("\n=== Reglas de derivación ===")
print("d/dx [x^n]   = n*x^(n-1)")
print("d/dx [e^x]   = e^x")
print("d/dx [log x] = 1/x")
print("d/dx [c]     = 0  (constante)")

🔗 Regla de la Cadena

d/dx[f(g(x))] = f'(g(x)) · g'(x). Es el corazón de backpropagation. Las capas de una red son funciones compuestas — la cadena las conecta.

import numpy as np

# Cálculo manual de backpropagation — 1 neurona
# y = sigmoid(w*x + b),  L = (y - target)^2

def forward_and_backward(x, w, b, target):
    """
    Forward + backward pass para 1 neurona.
    z = w*x + b
    a = sigmoid(z)
    L = 0.5 * (a - target)^2
    """
    # === FORWARD PASS ===
    z = w * x + b
    a = 1 / (1 + np.exp(-z))      # sigmoid
    L = 0.5 * (a - target) ** 2   # MSE simplificado

    # === BACKWARD PASS (cadena) ===
    # dL/da = (a - target)
    dL_da = a - target

    # da/dz = sigmoid'(z) = a*(1-a)
    da_dz = a * (1 - a)

    # dz/dw = x,  dz/db = 1

    # Regla de la cadena
    dL_dw = dL_da * da_dz * x  # dL/da * da/dz * dz/dw
    dL_db = dL_da * da_dz * 1  # dL/da * da/dz * dz/db

    return L, dL_dw, dL_db

# Entrenamiento manual 1 neurona
x, target = 2.0, 0.8
w, b = 0.5, 0.1
lr = 0.1

for epoch in range(10):
    L, dw, db = forward_and_backward(x, w, b, target)
    w -= lr * dw
    b -= lr * db
    if epoch == 0 or epoch == 9:
        print(f"Epoch {epoch}: L={L:.6f}, w={w:.4f}, b={b:.4f}")
▶ Output
Epoch 0: L=0.006243, w=0.5089, b=0.1044
Epoch 9: L=0.000041, w=0.5723, b=0.1362

∇ Gradientes y el Gradiente Vectorial

Con múltiples parámetros, la derivada se convierte en el gradiente: un vector que apunta en la dirección de máximo ascenso. Para descender, vamos en la dirección opuesta.

import numpy as np

# ── Gradiente de función multivariable ────────────
# f(x, y) = x^2 + y^2  (paraboloide — mínimo en origen)
# ∇f = [∂f/∂x, ∂f/∂y] = [2x, 2y]

def f(params):
    x, y = params
    return x**2 + y**2

def grad_f(params):
    x, y = params
    return np.array([2*x, 2*y])

# Visualización del paisaje de pérdidas
point = np.array([3.0, 4.0])
g = grad_f(point)
print(f"f({point}) = {f(point)}")
print(f"∇f({point}) = {g}  (apunta al máximo)")
print(f"-∇f({point}) = {-g}  (dirección de descenso)")
print(f"||∇f|| = {np.linalg.norm(g):.3f}  (magnitud del gradiente)")

# ── Gradiente numérico (gradient checking) ────────
def grad_numerico(f, params, h=1e-5):
    """Útil para verificar backprop implementado a mano."""
    grad = np.zeros_like(params, dtype=float)
    for i in range(len(params)):
    	p_plus  = params.copy(); p_plus[i]  += h
    	p_minus = params.copy(); p_minus[i] -= h
    	grad[i] = (f(p_plus) - f(p_minus)) / (2 * h)
    return grad

grad_anal = grad_f(point)
grad_num  = grad_numerico(f, point.astype(float))
error_rel = np.linalg.norm(grad_anal - grad_num) / (np.linalg.norm(grad_anal) + 1e-8)
print(f"\nGradient check error relativo: {error_rel:.2e}  ({'OK ✓' if error_rel < 1e-4 else 'FALLO ✗'})")
▶ Output

∇f([3. 4.]) = [6. 8.]
-∇f([3. 4.]) = [-6. -8.]
||∇f|| = 10.000
Gradient check error relativo: 1.33e-10  (OK ✓)

⛰️ Gradient Descent

θ ← θ - α · ∇L(θ). Actualiza parámetros en la dirección opuesta al gradiente. Tres variantes según cuántos datos usas para calcular el gradiente.

import numpy as np

np.random.seed(42)
# Dataset de ejemplo
X = np.random.randn(1000, 3)
y = X @ np.array([2.0, -1.0, 0.5]) + np.random.randn(1000) * 0.1

def mse_grad(X_batch, y_batch, w):
    y_pred = X_batch @ w
    error  = y_pred - y_batch
    loss   = np.mean(error**2) / 2
    grad   = X_batch.T @ error / len(y_batch)
    return loss, grad

# ── Batch GD: usa todos los datos ─────────────────
w_bgd = np.zeros(3)
for epoch in range(100):
    loss, grad = mse_grad(X, y, w_bgd)
    w_bgd -= 0.01 * grad
print(f"BGD: w={w_bgd.round(3)}, loss={loss:.6f}")

# ── SGD: 1 muestra a la vez ───────────────────────
w_sgd = np.zeros(3)
for epoch in range(10):
    idx = np.random.permutation(1000)
    for i in idx:
        loss, grad = mse_grad(X[[i]], y[[i]], w_sgd)
        w_sgd -= 0.01 * grad
print(f"SGD: w={w_sgd.round(3)}")

# ── Mini-batch GD (el más usado) ──────────────────
w_mgd = np.zeros(3)
batch_size = 32
for epoch in range(10):
    idx = np.random.permutation(1000)
    for start in range(0, 1000, batch_size):
        b = idx[start:start+batch_size]
        loss, grad = mse_grad(X[b], y[b], w_mgd)
        w_mgd -= 0.01 * grad
print(f"Mini-batch: w={w_mgd.round(3)}")
▶ Output
BGD:       w=[ 2.001 -1.000  0.499], loss=0.004989
SGD:       w=[ 1.998 -0.999  0.501]
Mini-batch w=[ 2.000 -1.000  0.500]

📉 Learning Rate: El Hiperparámetro Más Crítico

Demasiado grande → diverge. Demasiado pequeño → tarda siglos. La tasa de aprendizaje (α) controla el tamaño del paso y es el hiperparámetro con mayor impacto en el entrenamiento.

import numpy as np

# ── Schedulers de Learning Rate ───────────────────
class StepLR:
    def __init__(self, lr0, step_size, gamma=0.1):
        self.lr0, self.step, self.γ = lr0, step_size, gamma
    def get_lr(self, epoch):
        return self.lr0 * (self.γ ** (epoch // self.step))

class CosineAnnealingLR:
    def __init__(self, lr0, T_max):
        self.lr0, self.T = lr0, T_max
    def get_lr(self, epoch):
        return self.lr0 * 0.5 * (1 + np.cos(np.pi * epoch / self.T))

class WarmupCosineLR:
    def __init__(self, lr0, warmup_epochs, T_max):
        self.lr0, self.warmup, self.T = lr0, warmup_epochs, T_max
    def get_lr(self, epoch):
        if epoch < self.warmup:
            return self.lr0 * epoch / self.warmup
        t = epoch - self.warmup
        return self.lr0 * 0.5 * (1 + np.cos(np.pi * t / (self.T - self.warmup)))

schedulers = [
    ('Step LR',         StepLR(0.1, 30)),
    ('Cosine Anneal',   CosineAnnealingLR(0.1, 100)),
    ('Warmup + Cosine', WarmupCosineLR(0.1, 10, 100)),
]

for name, sched in schedulers:
    lrs = [sched.get_lr(e) for e in [0, 10, 30, 50, 99]]
    print(f"{name:20s}: {[f'{lr:.4f}' for lr in lrs]}")
▶ Output
Step LR             : ['0.1000', '0.1000', '0.0100', '0.0010', '0.0001']
Cosine Anneal       : ['0.1000', '0.0951', '0.0655', '0.0050', '0.0000']
Warmup + Cosine     : ['0.0000', '0.1000', '0.0745', '0.0058', '0.0000']
💡 Regla práctica: Empieza con lr=1e-3 (Adam) o lr=0.01 (SGD). Usa warmup para transformers. Cosine annealing para la mayoría de CNNs. Learning rate finder (fastai) para encontrar el valor óptimo.
