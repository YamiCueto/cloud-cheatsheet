⏱️ Notación Big O

Big O describe cómo crece el tiempo/espacio de un algoritmo en función del tamaño de entrada n. No mide segundos — mide la tasa de crecimiento en el peor caso.

# ── Comparación práctica ──────────────────────────
import time

n = 1_000_000
lista = list(range(n))
diccionario = {i: i for i in range(n)}

# O(n): buscar en lista — escanea todo
t0 = time.perf_counter()
resultado = 999_999 in lista
print(f"O(n) lista:      {(time.perf_counter()-t0)*1000:.2f}ms")

# O(1): buscar en dict — acceso directo
t0 = time.perf_counter()
resultado = 999_999 in diccionario
print(f"O(1) diccionario: {(time.perf_counter()-t0)*1000:.4f}ms")
▶ Output
O(n) lista:       12.34ms
O(1) diccionario:  0.0012ms
💡 Regla práctica: Para n = 10⁶ datos: O(1) = instantáneo, O(log n) = ~20 pasos, O(n) = 1 millón de pasos, O(n²) = 10¹² pasos (¡horas!).

📦 Arrays y HashMaps

Arrays = acceso O(1) por índice. HashMaps (dict en Python) = búsqueda e inserción O(1) amortizado. Son las estructuras más usadas en implementaciones de ML.

from collections import defaultdict, OrderedDict, Counter

# ── HashMap: Vocabulario NLP ─────────────────────
corpus = ["el gato come", "el perro duerme", "el gato duerme"]

vocab: dict[str, int] = {}
for texto in corpus:
    for palabra in texto.split():
        if palabra not in vocab:
            vocab[palabra] = len(vocab)  # índice único

print("Vocabulario:", vocab)
# {'el': 0, 'gato': 1, 'come': 2, 'perro': 3, 'duerme': 4}

# Encode de oraciones (Bag of Words)
def encode(texto, vocab):
    bow = [0] * len(vocab)
    for palabra in texto.split():
        if palabra in vocab:
            bow[vocab[palabra]] = 1
    return bow

print(encode("el gato duerme", vocab))
# [1, 1, 0, 0, 1]

# ── defaultdict: contar coocurrencias ────────────
coocurrencias = defaultdict(Counter)
for texto in corpus:
    palabras = texto.split()
    for i, p1 in enumerate(palabras):
        for p2 in palabras[i+1:]:
            coocurrencias[p1][p2] += 1

print(dict(coocurrencias['gato']))
▶ Output
Vocabulario: {'el': 0, 'gato': 1, 'come': 2, 'perro': 3, 'duerme': 4}
[1, 1, 0, 0, 1]
{'come': 1, 'duerme': 1}
dict / defaultdict

Vocabularios, caches, configuraciones. O(1) promedio.

set

Stop-words lookup, deduplicación rápida. O(1) contain.

list / deque

Buffers, sliding windows, BFS queue. O(1) append.

heapq

Top-K resultados, priority scheduling. O(log n) push/pop.

🔍 Algoritmos de Búsqueda

Buscar es fundamental: desde encontrar hiperparámetros óptimos (grid search) hasta recuperar los K documentos más similares (ANN search).

import bisect

# ── Binary Search: O(log n) ──────────────────────
def binary_search(arr, target):
    low, high = 0, len(arr) - 1
    while low <= high:
        mid = (low + high) // 2
        if   arr[mid] == target: return mid
        elif arr[mid] < target:  low  = mid + 1
        else:                    high = mid - 1
    return -1

# bisect: binary search + insert en O(log n)
umbrales = [0.1, 0.3, 0.5, 0.7, 0.9]
nuevo = 0.45
pos = bisect.bisect_left(umbrales, nuevo)
print(f"Insertar {nuevo} en posición {pos}")  # 2

# ── Grid Search: búsqueda de hiperparámetros ─────
import itertools

param_grid = {
    'learning_rate': [0.001, 0.01, 0.1],
    'n_layers':      [2, 4, 8],
    'dropout':       [0.1, 0.3],
}
keys = list(param_grid.keys())
combinaciones = list(itertools.product(*param_grid.values()))
print(f"Total combinaciones: {len(combinaciones)}")

for combo in combinaciones[:3]:
    params = dict(zip(keys, combo))
    print(params)

🧩 Patrones DSA en ML

Los mismos patrones de DSA aparecen constantemente en implementaciones de ML. Reconocerlos te hace más eficiente.

from collections import deque
import heapq

# ── Sliding Window: media móvil para time-series ──
def moving_average(datos, ventana):
    result = []
    window = deque()
    total = 0
    for x in datos:
        window.append(x)
        total += x
        if len(window) > ventana:
            total -= window.popleft()
        if len(window) == ventana:
            result.append(total / ventana)
    return result

serie = [1, 3, 5, 7, 9, 11, 13]
print("Media móvil k=3:", moving_average(serie, 3))
# [3.0, 5.0, 7.0, 9.0, 11.0]

# ── Two Pointers: detección de anomalías ─────────
def detectar_anomalias(datos, umbral=2.0):
    """Busca pares con diferencia > umbral en O(n)."""
    datos_ord = sorted(datos)
    anomalias = []
    i, j = 0, len(datos_ord) - 1
    while i < j:
        diff = datos_ord[j] - datos_ord[i]
        if diff > umbral:
            anomalias.append((datos_ord[i], datos_ord[j]))
            j -= 1
        else:
            i += 1
    return anomalias

scores = [0.1, 0.9, 0.2, 0.85, 0.15, 0.92]
print("Anomalías:", detectar_anomalias(scores, umbral=0.7))
▶ Output
Media móvil k=3: [3.0, 5.0, 7.0, 9.0, 11.0]
Anomalías: [(0.1, 0.92), (0.15, 0.9), (0.2, 0.85)]
